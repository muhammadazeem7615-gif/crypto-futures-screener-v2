//@version=5 indicator("VWMA 20 / EMA 9 MTF Screener [Binance Futures]", overlay=true) // Inputs src = input.source(close, "Source") vwmaLen = input.int(20, "VWMA Length") emaLen = input.int(9, "EMA Length") // Function to calculate cross status for a given timeframe f_checkCrossover(tf) => r_vwma = request.security(syminfo.tickerid, tf, ta.vwma(src, vwmaLen), barmerge.gaps_off, barmerge.lookahead_off) r_ema = request.security(syminfo.tickerid, tf, ta.ema(src, emaLen), barmerge.gaps_off, barmerge.lookahead_off) isBullCross = ta.crossover(r_ema, r_vwma) isBearCross = ta.crossunder(r_ema, r_vwma) isBullish = r_ema > r_vwma [isBullish, isBullCross, isBearCross] // Check across 1H, 2H, 4H, and 1D [bull1h, crossUp1h, crossDown1h] = f_checkCrossover("60") [bull2h, crossUp2h, crossDown2h] = f_checkCrossover("120") [bull4h, crossUp4h, crossDown4h] = f_checkCrossover("240") [bull1d, crossUp1d, crossDown1d] = f_checkCrossover("D") // Create a Table Dashboard on the Chart var table dash = table.new(position = position.top_right, columns = 5, rows = 2, bgcolor = color.new(color.black, 20), border_width = 1, border_color = color.gray) if barstate.islast // Header Row table.cell(dash, 0, 0, "Timeframe", text_color=color.white, text_size=size.small) table.cell(dash, 1, 0, "1H", text_color=color.white, text_size=size.small) table.cell(dash, 2, 0, "2H", text_color=color.white, text_size=size.small) table.cell(dash, 3, 0, "4H", text_color=color.white, text_size=size.small) table.cell(dash, 4, 0, "1D", text_color=color.white, text_size=size.small) // Status Row table.cell(dash, 0, 1, "Status", text_color=color.white, text_size=size.small) // Inline rendering logic to avoid scope/syntax compilation bugs f_render(col, bullish, cUp, cDown) => txt = bullish ? "BULLISH" : "BEARISH" if cUp txt := "🟢 CROSS UP" if cDown txt := "🔴 CROSS DN" bg = cUp ? color.green : cDown ? color.red : (bullish ? color.new(color.green, 70) : color.new(color.red, 70)) table.cell(dash, col, 1, txt, text_color=color.white, bgcolor=bg, text_size=size.small) f_render(1, bull1h, crossUp1h, crossDown1h) f_render(2, bull2h, crossUp2h, crossDown2h) f_render(3, bull4h, crossUp4h, crossDown4h) f_render(4, bull1d, crossUp1d, crossDown1d)
